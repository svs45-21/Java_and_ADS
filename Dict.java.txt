import java.util.ArrayList;
 public class Dict{
     public static void main(String[] args){
         DictArray D = new DictArray();
         D.Insert(101,"Surya");
         D.Insert(102,"Reka");
         D.Insert(103,"Rahul");
         System.out.println("Original Array");
         D.Display();
         String Result = D.Search(103);
         System.out.println("Result of search: " +Result);
         D.Delete(103);
         System.out.println("Array after the deletion");
         D.Display();
         D.Insert(101,"Shree");
         System.out.println("Array after update");
         D.Display();
     }
     
 }
 
 class Pair{
     int rollNumber;
     String Name;
     Pair(int rollNumber, String Name){
         this.rollNumber = rollNumber;
         this.Name = Name;
     }
 }
 
 class DictArray
 {
     ArrayList<Pair> obj = new ArrayList<>();
     
    //Insert or Update
    void Insert(int rollNumber, String Name){
        for (Pair element : obj){
            if (element.rollNumber == rollNumber){
                element.Name = Name; 
                return;
            }
        }
        
     obj.add(new Pair(rollNumber,Name));
    }
    //Display
    void Display(){
        for(Pair element : obj){
            System.out.println("roll number = "+element.rollNumber+ "student Name = "+ element.Name);
        }
    }
    
    //Search
    String Search(int rollNumber){
        for (Pair element :obj){
            if (element.rollNumber == rollNumber){
                return element.Name;
            }
        }
        return "Student Name not found";
    }
    
    //Delete
    void Delete(int rollNumber){
        obj.removeIf(o -> o.rollNumber == rollNumber);
    }
 }
